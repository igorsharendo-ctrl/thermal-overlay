cmake_minimum_required(VERSION 3.20)
project(ThermalOverlay LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "ThermalOverlay requires Windows")
endif()
add_executable(ThermalOverlay WIN32 src/main.cpp)
set_target_properties(ThermalOverlay PROPERTIES OUTPUT_NAME Thermal)
target_compile_features(ThermalOverlay PRIVATE cxx_std_17)
target_link_libraries(ThermalOverlay PRIVATE advapi32 user32 gdi32 comdlg32 shell32 ole32 oleaut32 wbemuuid pdh dxgi uuid)
if(MSVC)
    set_property(TARGET ThermalOverlay PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(ThermalOverlay PRIVATE /W4 /utf-8)
else()
    target_compile_options(ThermalOverlay PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(ThermalOverlay PRIVATE -static -municode -Wl,--no-insert-timestamp)
endif()

add_executable(ThermalOverlayCpuService WIN32 src/cpu_service.cpp)
target_compile_features(ThermalOverlayCpuService PRIVATE cxx_std_17)
target_link_libraries(ThermalOverlayCpuService PRIVATE advapi32 user32)
if(MSVC)
    set_property(TARGET ThermalOverlayCpuService PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(ThermalOverlayCpuService PRIVATE /W4 /utf-8)
else()
    target_compile_options(ThermalOverlayCpuService PRIVATE -Wall -Wextra)
    target_link_options(ThermalOverlayCpuService PRIVATE -static -municode -Wl,--no-insert-timestamp)
endif()
add_dependencies(ThermalOverlay ThermalOverlayCpuService)
add_custom_command(TARGET ThermalOverlay POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory $<TARGET_FILE_DIR:ThermalOverlay>/support
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:ThermalOverlayCpuService> $<TARGET_FILE_DIR:ThermalOverlay>/support/ThermalOverlayCpuService.exe
    COMMAND ${CMAKE_COMMAND} -E copy_if_different ${CMAKE_SOURCE_DIR}/setup-cpu.ps1 $<TARGET_FILE_DIR:ThermalOverlay>/support/setup-cpu.ps1
    COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/third_party/pawnio/modules $<TARGET_FILE_DIR:ThermalOverlay>/support/modules)
# build.ps1 additionally packages the verified official PawnIO installer.
include(CTest)
if(BUILD_TESTING)
    add_executable(cpu_sensors_test tests/cpu_sensors_test.cpp)
    target_compile_features(cpu_sensors_test PRIVATE cxx_std_17)
    target_link_libraries(cpu_sensors_test PRIVATE advapi32)
    add_test(NAME cpu_sensors_test COMMAND cpu_sensors_test)
endif()

if(BUILD_TESTING)
    add_executable(cpu_setup_status_test tests/cpu_setup_status_test.cpp)
    target_compile_features(cpu_setup_status_test PRIVATE cxx_std_17)
    target_link_libraries(cpu_setup_status_test PRIVATE advapi32 user32 gdi32 comdlg32 shell32 ole32 oleaut32 wbemuuid pdh dxgi uuid)
    if(MSVC)
        target_compile_options(cpu_setup_status_test PRIVATE /utf-8)
    else()
        target_link_options(cpu_setup_status_test PRIVATE -static -municode)
    endif()
    add_test(NAME cpu_setup_status_test COMMAND cpu_setup_status_test)
endif()
